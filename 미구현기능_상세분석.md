### 2.3 Health Check 개선 ⚠️
**현황**: `/health` 엔드포인트만 있음 (DB/Redis 체크 없음)

**해야 할 일**:

```python
# backend/src/api/health.py
from fastapi import APIRouter, Depends, status
from sqlalchemy.ext.asyncio import AsyncSession
from src.database.connection import get_session
import redis.asyncio as redis

router = APIRouter()

@router.get("/health/live")
async def liveness():
    """Liveness Probe - Pod 재시작 여부 판단"""
    return {"status": "ok"}

@router.get("/health/ready")
async def readiness(session: AsyncSession = Depends(get_session)):
    """Readiness Probe - 트래픽 수신 가능 여부"""
    checks = {
        "database": False,
        "redis_cache": False,
        "redis_queue": False,
    }

    # 1. DB 연결 체크
    try:
        await session.execute(text("SELECT 1"))
        checks["database"] = True
    except Exception as e:
        pass

    # 2. Redis Cache 체크
    try:
        r = redis.from_url(os.getenv("REDIS_CACHE_URL"))
        await r.ping()
        checks["redis_cache"] = True
    except:
        pass

    # 3. Redis Queue 체크
    try:
        r = redis.from_url(os.getenv("REDIS_QUEUE_URL"))
        await r.ping()
        checks["redis_queue"] = True
    except:
        pass

    all_healthy = all(checks.values())
    status_code = status.HTTP_200_OK if all_healthy else status.HTTP_503_SERVICE_UNAVAILABLE

    return {"status": "healthy" if all_healthy else "unhealthy", "checks": checks}
```

**K8s Deployment에 적용**:
```yaml

    livenessProbe:

      periodSeconds: 10

      httpGet:
```

### 3.3 CSRF 보호 강화 ⚠️
**현황**: CSRF 토큰 생성 있지만 검증 부족

**해야 할 일**:

```python
# backend/src/middleware/csrf.py
from fastapi import Request, HTTPException
import hmac
import hashlib

class CSRFMiddleware:
    async def __call__(self, request: Request, call_next):
        if request.method in ["POST", "PUT", "DELETE", "PATCH"]:
            # 1. 쿠키에서 CSRF 토큰 읽기
            cookie_token = request.cookies.get("csrf_token")
            # 2. 헤더에서 CSRF 토큰 읽기
            header_token = request.headers.get("X-CSRF-Token")

            if not cookie_token or not header_token:
                raise HTTPException(status_code=403, detail="CSRF token missing")

            # 3. 토큰 검증 (Double Submit Cookie)
            if not hmac.compare_digest(cookie_token, header_token):
                raise HTTPException(status_code=403, detail="CSRF token invalid")

        return await call_next(request)
```

### 3.7 비밀번호 정책 강화 ❌
**현황**: 비밀번호 복잡도 검증 없음

**해야 할 일**:

```python
# backend/src/services/auth_service.py
import re
import requests

def validate_password(password: str) -> tuple[bool, str]:
    """
    비밀번호 정책 검증
    - 최소 8자, 최대 64자
    - 영문 대소문자, 숫자, 특수문자 포함
    - 연속된 문자 3개 이상 금지
    - Pwned Passwords API 체크
    """
    # 1. 길이 검증
    if len(password) < 8 or len(password) > 64:
        return False, "비밀번호는 8자 이상 64자 이하여야 합니다."

    # 2. 복잡도 검증
    if not re.search(r'[A-Z]', password):
        return False, "대문자를 포함해야 합니다."
    if not re.search(r'[a-z]', password):
        return False, "소문자를 포함해야 합니다."
    if not re.search(r'[0-9]', password):
        return False, "숫자를 포함해야 합니다."
    if not re.search(r'[!@#$%^&*(),.?":{}|<>]', password):
        return False, "특수문자를 포함해야 합니다."

    # 3. 연속 문자 금지
    for i in range(len(password) - 2):
        if password[i] == password[i+1] == password[i+2]:
            return False, "동일한 문자를 3번 연속 사용할 수 없습니다."

    # 4. Pwned Passwords API 체크
    sha1_hash = hashlib.sha1(password.encode()).hexdigest().upper()
    prefix = sha1_hash[:5]
    suffix = sha1_hash[5:]

    response = requests.get(f"https://api.pwnedpasswords.com/range/{prefix}")
    if suffix in response.text:
        return False, "이 비밀번호는 유출된 적이 있습니다. 다른 비밀번호를 사용하세요."

    return True, "OK"
```

```python
import httpx
import os

class GeocodingService:
    def __init__(self):
        self.kakao_api_key = os.getenv("KAKAO_REST_API_KEY")

    async def geocode_address(self, address: str) -> tuple[float, float]:
        """주소 → 위도/경도 변환"""
        async with httpx.AsyncClient() as client:
            response = await client.get(
                "https://dapi.kakao.com/v2/local/search/address.json",
                headers={"Authorization": f"KakaoAK {self.kakao_api_key}"},
                params={"query": address}
            )
            data = response.json()

            if data.get("documents"):
                doc = data["documents"][0]
                return float(doc["y"]), float(doc["x"])  # lat, lng

            return None, None
```

**현황**: 168건 중 1건만 매칭 (0.6%)

**문제 분석**:
1. 소규모 화재는 뉴스화 안 됨
2. Embedding 유사도 임계값이 너무 높음
3. 시간 범위가 좁음

**해야 할 일**:

```python
# backend/src/services/matching/news_matcher.py

class NewsMatcherService:
    def __init__(self):
        self.min_similarity = 0.65  # 0.8 → 0.65로 낮춤
        self.time_window_hours = 48  # 24 → 48시간으로 확대

    async def match_news_with_incident(self, incident: FireIncident):
        """뉴스 매칭 개선"""
        # 1. 키워드 기반 1차 필터링
        keywords = self._extract_keywords(incident)
        candidate_news = await self._search_by_keywords(keywords)

        # 2. Embedding 유사도 계산
        incident_embedding = await self.embed_model.embed(incident.title + " " + incident.location_address)

        matches = []
        for news in candidate_news:
            news_embedding = await self.embed_model.embed(news.title + " " + news.content)
            similarity = cosine_similarity(incident_embedding, news_embedding)

            if similarity >= self.min_similarity:
                matches.append((news, similarity))

        # 3. 시간 필터링 (더 넓게)
        time_filtered = [
            (news, score) for news, score in matches
            if abs((news.published_at - incident.occurred_at).total_seconds()) < self.time_window_hours * 3600
        ]

        return sorted(time_filtered, key=lambda x: x[1], reverse=True)[:5]

    def _extract_keywords(self, incident: FireIncident) -> List[str]:
        """키워드 추출"""
        # 지역명
        location_match = re.search(r'(서울|경기|인천|부산|대구|광주|대전|울산|세종|제주)', incident.location_address)
        location = location_match.group(1) if location_match else ""

        # 소방서명 (ex: 강남소방서 → 강남)
        station_name = incident.title.replace("소방서", "")

        return [
            "화재",
            location,
            station_name,
            "소방",
        ]

    async def _search_by_keywords(self, keywords: List[str]) -> List[NewsContent]:
        """키워드로 뉴스 검색"""
        query = " ".join(keywords)

        # Naver News API
        response = await self.naver_client.get(
            "https://openapi.naver.com/v1/search/news.json",
            params={
                "query": query,
                "display": 100,
                "sort": "date",
            }
        )

        return response.json()["items"]
```

```python
from fastapi import APIRouter, Depends, HTTPException
import httpx
import os

router = APIRouter()

@router.post("/refunds")
async def request_refund(
    donation_id: UUID,
    reason: str,
    session: AsyncSession = Depends(get_session),
    current_user: User = Depends(get_current_user)
):
    """환불 요청"""
    # 1. 기부 내역 조회
    donation = await session.get(Donation, donation_id)
    if not donation:
        raise HTTPException(status_code=404, detail="Donation not found")

    if donation.user_id != current_user.id:
        raise HTTPException(status_code=403, detail="Forbidden")

    # 2. 환불 가능 여부 체크
    if donation.status == "refunded":
        raise HTTPException(status_code=400, detail="Already refunded")

    # 환불 기한 체크 (7일 이내)
    if (datetime.now(timezone.utc) - donation.created_at).days > 7:
        raise HTTPException(status_code=400, detail="Refund period expired")

    # 3. Toss Payments 환불 API 호출
    toss_response = await httpx.post(
        f"https://api.tosspayments.com/v1/payments/{donation.payment_key}/cancel",
        json={
            "cancelReason": reason,
            "cancelAmount": donation.amount,
        },
        headers={
            "Authorization": f"Basic {base64.b64encode(f'{os.getenv(\"TOSS_SECRET_KEY\")}:'.encode()).decode()}",
            "Content-Type": "application/json",
        },
    )

    if toss_response.status_code != 200:
        raise HTTPException(status_code=500, detail="Refund failed")

    # 4. 환불 기록 저장
    donation.status = "refunded"
    donation.refunded_at = datetime.now(timezone.utc)
    donation.refund_reason = reason

    refund = Refund(
        donation_id=donation.id,
        amount=donation.amount,
        reason=reason,
        processed_at=datetime.now(timezone.utc),
    )
    session.add(refund)

    await session.commit()

    return {"refund_id": refund.id, "status": "completed"}
```

```typescript
import { Component, ReactNode } from 'react';
import * as Sentry from '@sentry/nextjs';

interface Props {
  children: ReactNode;
  fallback?: ReactNode;
}

interface State {
  hasError: boolean;
  error?: Error;
}

export class ErrorBoundary extends Component<Props, State> {
  constructor(props: Props) {
    super(props);
    this.state = { hasError: false };
  }

  static getDerivedStateFromError(error: Error): State {
    return { hasError: true, error };
  }

  componentDidCatch(error: Error, errorInfo: any) {
    // Sentry에 에러 전송
    Sentry.captureException(error, { extra: errorInfo });
    console.error('ErrorBoundary caught:', error, errorInfo);
  }

  render() {
    if (this.state.hasError) {
      return this.props.fallback || (
        <div className="flex items-center justify-center min-h-screen">
          <div className="text-center">
            <h1 className="text-2xl font-bold text-red-600 mb-4">
              오류가 발생했습니다
            </h1>
            <p className="text-gray-600 mb-4">
              {this.state.error?.message || '알 수 없는 오류'}
            </p>
            <button
              onClick={() => window.location.reload()}
              className="px-4 py-2 bg-blue-600 text-white rounded"
            >
              페이지 새로고침
            </button>
          </div>
        </div>
      );
    }

    return this.props.children;
  }
}
```

```python
# backend/src/cache/decorator.py
from functools import wraps
import json
import hashlib

def cache_result(ttl: int = 300):
    """API 응답 캐싱 데코레이터"""
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # 캐시 키 생성
            cache_key = f"{func.__name__}:{hashlib.md5(json.dumps(kwargs, sort_keys=True).encode()).hexdigest()}"

            # 캐시 조회
            cached = await redis_client.get(cache_key)
            if cached:
                return json.loads(cached)

            # 함수 실행
            result = await func(*args, **kwargs)

            # 캐시 저장
            await redis_client.setex(cache_key, ttl, json.dumps(result))

            return result
        return wrapper
    return decorator

# 사용
@router.get("/api/stations")
@cache_result(ttl=300)  # 5분 캐싱
async def list_stations(...):
    ...

@router.get("/api/incidents")

async def list_incidents(...):
    ...
```
